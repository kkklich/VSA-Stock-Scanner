"""Paste into a native Python Code node in Run Once for Each Item mode.

The final return call is already included at the end of this file.
Input comes from Extract From File, with pdfBase64 as its destination field.
The Python runner supplies STOCKPILOT_XTB_STATEMENT_PASSWORD through its environment.
"""

import base64
import os
import re
import unicodedata
from datetime import datetime
from decimal import Decimal, InvalidOperation
from io import BytesIO
from uuid import NAMESPACE_URL, uuid5

from pypdf import PdfReader
from pypdf.errors import DependencyError, PyPdfError

MAX_BYTES = 5 * 1024 * 1024
MAX_PAGES = 40
MAX_ORDERS = 1000


def plain(value):
    value = value.replace("ł", "l").replace("Ł", "L")
    return unicodedata.normalize("NFKD", value).encode("ascii", "ignore").decode().lower()


def number(value, *, positive=False, absolute=False):
    cleaned = re.sub(r"\s+", "", value).replace("\u2212", "-")
    if not re.fullmatch(r"[+-]?\d+(?:[,.]\d+)?", cleaned):
        raise ValueError("A numeric cell could not be read reliably.")
    result = Decimal(cleaned.replace(",", "."))
    if absolute:
        result = abs(result)
    if result < 0 or (positive and result == 0):
        raise ValueError("A transaction contains an invalid quantity, price or fee.")
    if result > Decimal("1000000000000") or result.as_tuple().exponent < -8:
        raise ValueError("A transaction exceeds the portfolio API's numeric limits.")
    return result


def header_columns(line):
    labels = {
        re.sub(r"\s+", " ", plain(part.group().strip())): part.start()
        for part in re.finditer(r"\S(?:.*?\S)?(?=\s{2,}|$)", line)
    }
    if "symbol, ktorego" in labels:
        labels["symbol"] = labels["symbol, ktorego"]
    required = {
        "symbol": ("symbol", "nazwa"),
        "quantity": ("wolumen", "data i czas"),
        "time": ("data i czas", "zlecenie"),
        "price": ("cena", "cena calkowita"),
        "asset": ("klasa aktywow", "waluta"),
        "currency": ("waluta", "kurs"),
        "fx": ("koszty", "prowizja"),
        "fees": ("prowizja", "laczne koszty"),
    }
    missing = sorted({label for pair in required.values() for label in pair} - labels.keys())
    if missing:
        raise ValueError("Unrecognized transaction columns: " + ", ".join(missing))
    columns = {key: slice(labels[start], labels[end]) for key, (start, end) in required.items()}
    if any(column.start >= column.stop for column in columns.values()):
        raise ValueError("The transaction columns overlap in the extracted PDF text.")
    return columns


def account_summary(lines):
    pattern = re.compile(r"\b(?:konto|numer rachunku)\s{2,}(waluta rachunku)\s{2,}(saldo)")
    for index, line in enumerate(lines[:-1]):
        match = pattern.search(plain(line))
        if match:
            values = lines[index + 1]
            account = values[match.start() : match.start(1)].strip()
            currency = values[match.start(1) : match.start(2)].strip()
            if not re.fullmatch(r"\d{1,32}", account):
                raise ValueError("The XTB account number could not be read reliably.")
            if not re.fullmatch(r"[A-Z]{3}", currency):
                raise ValueError("The account currency could not be read reliably.")
            return account, currency
    return None


def transaction_row(order, line, continuation, columns, report_date, account_currency, side):
    values = {key: line[column].strip() for key, column in columns.items()}
    time_continuation = continuation[columns["time"]].strip()
    if re.fullmatch(r"\d{2}:\d{2}:\d{2}", time_continuation):
        values["time"] += " " + time_continuation
    asset_continuation = continuation[columns["asset"]].strip()
    if plain(asset_continuation).startswith("dot"):
        values["asset"] += " " + asset_continuation
    asset = re.sub(r"[^a-z]", "", plain(values["asset"]))
    if asset not in {"akcja", "akcje", "prawaulamkowedotakcji", "prawoulamkowedotakcji"}:
        raise ValueError("An asset type requires manual review: " + values["asset"])

    executed = datetime.strptime(values["time"], "%d.%m.%Y %H:%M:%S")
    if executed.date() != report_date:
        raise ValueError("A transaction date differs from the report date.")
    symbol = values["symbol"].upper()
    if not re.fullmatch(r"[A-Z0-9][A-Z0-9.\-]{0,29}", symbol):
        raise ValueError("A stock symbol could not be read reliably.")
    ticker = symbol.lower()
    for suffix, replacement in {
        ".pl": "",
        ".wa": "",
        ".fr": ".pa",
        ".nl": ".as",
        ".uk": ".l",
    }.items():
        if ticker.endswith(suffix):
            ticker = ticker[: -len(suffix)] + replacement
            break

    currency = values["currency"]
    if currency not in {"PLN", "USD", "EUR", "GBP", "GBp"}:
        raise ValueError("Unsupported or unreadable stock currency: " + currency)
    fees = number(values["fees"], absolute=True)
    if fees and currency != account_currency:
        raise ValueError("The fee currency differs from the stock currency and needs review.")
    fx_cost = number(values["fx"], absolute=True) if values["fx"] else Decimal(0)
    reference = f"xtb:daily/{order}/{executed:%Y%m%dT%H%M%S}/{symbol}:{side}"
    return {
        "orderId": order,
        "sourceId": reference,
        "executedAt": executed.isoformat(timespec="microseconds"),
        "transaction": {
            "id": str(uuid5(NAMESPACE_URL, reference)),
            "ticker": ticker,
            "date": executed.date().isoformat(),
            "side": side,
            "quantity": format(number(values["quantity"], positive=True), "f"),
            "price": format(number(values["price"], positive=True), "f"),
            "fees": format(fees, "f"),
            "currency": currency,
        },
    }, bool(fx_cost)


def parse_pages(pages):
    page_lines = [[line for line in page.splitlines() if line.strip()] for page in pages]
    first_page = page_lines[0]
    title_text = re.sub(r"\s+", " ", plain(pages[0]))
    title = re.search(r"potwierdzenie zlecen wykonanych w dniu\s+(\d{2}\.\d{2}\.\d{4})", title_text)
    if title is None:
        raise ValueError("The expected Polish XTB DailyStatement title was not found.")
    report_date = datetime.strptime(title[1], "%d.%m.%Y").date()
    account = account_summary(first_page)
    if account is None:
        raise ValueError("The account-summary columns were not recognized.")
    account_id, account_currency = account
    transactions = []
    references = set()
    section = None
    remaining = 0
    total = 0
    fx_cost = False

    for lines in page_lines:
        repeated_account = account_summary(lines)
        if repeated_account is not None and repeated_account != account:
            raise ValueError("The report contains inconsistent XTB accounts.")
        columns = None
        for index, line in enumerate(lines):
            normalized = re.sub(r"\s+", " ", plain(line.strip()))
            if normalized.startswith("wykonane zlecenia"):
                match = re.fullmatch(
                    r"wykonane zlecenia (kupna|sprzedazy)\b.*?\(\s*(\d+)\s*\)", normalized
                )
                if match is None:
                    raise ValueError("An executed-order section heading was not recognized.")
                new_section = match.groups()
                if remaining and new_section != section:
                    raise ValueError("The preceding section has unread transaction rows.")
                if not remaining:
                    section = new_section
                    remaining = int(match[2])
                    total += remaining
                    if total > MAX_ORDERS:
                        raise ValueError(
                            "The report exceeds the existing 1,000-order import limit."
                        )
                columns = None
            elif section and re.match(r"nr numer zlecenia\b", normalized):
                columns = header_columns(line)
            elif section and (order := re.match(r"\s*\d+\s+(\d{1,24})(?=\s)", line)):
                if remaining == 0 or columns is None:
                    raise ValueError(
                        "A transaction row has missing columns or an unexpected count."
                    )
                following = lines[index + 1] if index + 1 < len(lines) else ""
                side = "buy" if section[0] == "kupna" else "sell"
                trade, has_fx = transaction_row(
                    order[1], line, following, columns, report_date, account_currency, side
                )
                if trade["sourceId"] in references:
                    raise ValueError("A repeated order/fill needs manual review.")
                references.add(trade["sourceId"])
                transactions.append(trade)
                remaining -= 1
                fx_cost |= has_fx
    if section is None or remaining:
        raise ValueError("The extracted transaction count does not match the report headings.")
    return {
        "status": "ready",
        "brokerAccountId": account_id,
        "brokerAccountCurrency": account_currency,
        "reportDate": report_date.isoformat(),
        "transactions": transactions,
        "warnings": ["fx_cost_not_imported"] if fx_cost else [],
    }


def transform_item(input_json):
    pages = []
    try:
        encoded = input_json.get("pdfBase64")
        if not isinstance(encoded, str) or not encoded:
            raise ValueError("Missing pdfBase64. Add Extract From File before this node.")
        if len(encoded) > ((MAX_BYTES + 2) // 3) * 4:
            raise ValueError("The PDF exceeds the existing 5 MiB import limit.")
        pdf_data = base64.b64decode(encoded, validate=True)
        if len(pdf_data) > MAX_BYTES or not pdf_data.startswith(b"%PDF-"):
            raise ValueError("The supplied data is not a supported PDF upload.")
        reader = PdfReader(BytesIO(pdf_data), strict=True)
        if reader.is_encrypted:
            password = os.environ.get("STOCKPILOT_XTB_STATEMENT_PASSWORD")
            if not password:
                raise ValueError("The PDF password is missing from the n8n Python runner.")
            if not reader.decrypt(password):
                raise ValueError("The PDF password is incorrect.")
        if not 1 <= len(reader.pages) <= MAX_PAGES:
            raise ValueError("The PDF page count is outside the existing import limits.")
        for page in reader.pages:
            text = page.extract_text(extraction_mode="layout") or ""
            if len(text) > 100_000:
                raise ValueError("An extracted page exceeds the existing import text limit.")
            pages.append(text)
        return parse_pages(pages)
    except (ValueError, InvalidOperation, PyPdfError, DependencyError, PermissionError) as error:
        return {
            "status": "needs_review",
            "error": str(error),
            "transactions": [],
            "pages": [{"page": index + 1, "text": text} for index, text in enumerate(pages)],
        }
